/* RENDERIZAÇÃO AUTOMÁTICA DOS CARDS COM DADOS DA PLANILHA */ function renderApp() { const row = document.getElementById("cardRow"); row.innerHTML = ""; const startIndex = currentPage * cardsPerRow; const pageItems = currentFiltered.slice(startIndex, startIndex + cardsPerRow); pageItems.forEach((can, localIndex) => { const globalIndex = startIndex + localIndex; // 1. Pega a foto da frente (testando várias propriedades possíveis do seu JSON) const rawFront = can.images ? (can.images.front || can.images.frente || can.images.cover || can.images.capa || can.brandLogo) : (can.front || can.frente || can.cover || can.capa || can.brandLogo || ""); const frontImg = getOptimizedImageUrl(rawFront, 400); // 2. Mapeamento flexível dos textos do rodapé const brandName = can.brewery || can.marca || can.cervejaria || can.name || can.nome || "Cerveja"; const countryName = can.country || can.pais || can.país || can.origem || ""; const volumeText = can.volume || can.ml || can.conteudo || ""; const flagUrl = getFlagUrlByCountry(countryName); // 3. Define a cor do rodapé const brandKey = String(brandName).toLowerCase().trim(); const cardColor = can.brandColor || can.color || can.cor || BRAND_COLORS[brandKey] || "#008234"; const card = document.createElement("div"); card.className = `card ${currentTab === 'sets' ? 'card-aspect-sets' : ''}`; card.tabIndex = 0; card.dataset.globalIndex = globalIndex; card.onmouseenter = () => { document.querySelectorAll(".card").forEach(c => c.classList.remove("focused")); card.classList.add("focused"); activeLocalIndex = localIndex; updateHeroPanel(can, globalIndex); }; card.onfocus = () => { if (document.getElementById("zoomModal").classList.contains("active") || document.getElementById("exitModal").classList.contains("active")) return; document.querySelectorAll(".card").forEach(c => c.classList.remove("focused")); card.classList.add("focused"); activeLocalIndex = localIndex; updateHeroPanel(can, globalIndex); }; card.onclick = (e) => { e.stopPropagation(); openZoomModal(0); }; card.onkeydown = (e) => { if (document.getElementById("zoomModal").classList.contains("active") || document.getElementById("exitModal").classList.contains("active")) return; if (e.key === "Enter" || e.key === " ") { e.preventDefault(); openZoomModal(0); } else if (e.key === "ArrowRight") { if (localIndex === pageItems.length - 1 && (startIndex + cardsPerRow) < currentFiltered.length) { e.preventDefault(); currentPage++; renderApp(); focusCardSafely(0); } } else if (e.key === "ArrowLeft") { if (localIndex === 0) { e.preventDefault(); if (currentPage > 0) { currentPage--; renderApp(); focusCardSafely(cardsPerRow - 1); } else { clearAllCardFocus(); openMenu(); } } } else if (e.key === "ArrowDown") { e.preventDefault(); if ((startIndex + cardsPerRow) < currentFiltered.length) { currentPage++; renderApp(); focusCardSafely(activeLocalIndex); } } else if (e.key === "ArrowUp") { e.preventDefault(); if (currentPage > 0) { currentPage--; renderApp(); focusCardSafely(activeLocalIndex); } } }; // Monta a foto + rodapé colorido exatamente como no layout enviado card.innerHTML = `
${brandName}
${flagUrl ? `` : ''} ${countryName}
${volumeText ? `
${volumeText}
` : ''}
`; row.appendChild(card); }); }
